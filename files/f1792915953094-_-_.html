구글 시트 실행 버전

구글 시트에 Apps Script를 붙여넣으면 시트 상단에 큐알장부 메뉴가 생기고, 데이터가 시트에 그대로 저장됩니다.

  • 전 기능 탑재 — 탭 10개 (전표 · 장부 · 재무제표 · 결산 · 가결산 · 부가세·세무 · 분석 · 업로드 · 기준정보 · 설정)
  • 데이터는 시트에 그대로 저장되고, 보고서는 [시트로 저장] 버튼으로 기록
  • 매출 전표 저장 시 거래명세서(TR-)·세금계산서(TX-) 자동 발행·연동
  • 크롬 버전과 백업 파일 공유 — [설정 → 백업 공유]에서 qr-backup.json 내려받기·복원

구글 시트 버전 설치 — 7단계

PC 브라우저에서 진행하세요. 약 3분이면 끝납니다.

  1. 새 구글 시트를 만듭니다 (sheets.new 입력).
  2. 상단 확장 프로그램 → Apps Script를 엽니다.
  3. 에디터의 기본 코드를 모두 지우고, 아래 Code.gs 코드를 붙여넣습니다.
  4. + → HTML로 새 파일을 만들고 이름을 정확히 UI로 지정한 뒤, 아래 UI.html 코드를 붙여넣습니다. (기존 내용 전체 교체)
  5. 저장(Ctrl+S) 후 시트 탭으로 돌아와 새로고침하면 상단에 큐알장부 메뉴가 생깁니다.
  6. 큐알장부 → 초기 설정 (시트 자동 생성)을 먼저 실행합니다. 첫 실행 시 Google 승인 화면이 나오면 [승인 필요] → 계정 선택 → [고급] → [안전하지 않은 페이지로 이동] → [허용]을 누릅니다. (직접 만든 스크립트라 안전합니다)
  7. 큐알장부 → 큐알장부 열기 (전체 화면)를 누르면 전 기능 화면이 열립니다 (탭 10개).

function sh_(name){ return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); }
function ensureSheet_(name){
  const ss=SpreadsheetApp.getActiveSpreadsheet();
  let s=ss.getSheetByName(name);
  if(!s){
    s=ss.insertSheet(name);
    s.getRange(1,1,1,HEADERS[name].length).setValues([HEADERS[name]])
      .setFontWeight('bold').setBackground('#e8eaed');
    s.setFrozenRows(1);
  }
  return s;
}
function str_(v){
  if(v instanceof Date) return Utilities.formatDate(v,Session.getScriptTimeZone(),'yyyy-MM-dd');
  return String(v==null?'':v).trim();
}
function num_(v){ const n=Number(String(v==null?0:v).toString().replace(/,/g,'')); return Number.isFinite(n)?n:0; }
function readRows_(name){
  const s=sh_(name); if(!s) return [];
  const v=s.getDataRange().getValues(), keys=HEADERS[name], out=[];
  for(let i=1;i<v.length;i++){
    const o={}; let empty=true;
    for(let j=0;j<keys.length;j++){ o[keys[j]]=v[i][j] instanceof Date?v[i][j]:v[i][j]; if(str_(v[i][j])!=='') empty=false; }
    if(!empty) o.__row=i+1;
    if(!empty) out.push(o);
  }
  return out;
}
function appendRows_(name,arr2d,fmtCols){
  if(!arr2d.length) return;
  const sh=ensureSheet_(name), start=sh.getLastRow()+1;
  sh.getRange(start,1,arr2d.length,arr2d[0].length).setValues(arr2d);
  if(fmtCols) fmtCols.forEach(c=>sh.getRange(start,c[0],arr2d.length,c[1]).setNumberFormat('#,##0'));
}

function importPreview(text){
  const map=amap_(), rows=[];
  String(text||'').split(/\r?\n/).forEach(line=>{
    if(!line.trim()) return;
    const c=line.indexOf('\t')>=0?line.split('\t'):line.split(',');
    if(c.length<5) return;
    const date=str_(c[0]), memo=str_(c[1]), acctTxt=str_(c[2]), p=str_(c[3]), d=num_(c[4]), cr=num_(c[5]);
    if(!date&&!d&&!cr) return;
    let code=0, how='';
    if(/^\d+$/.test(acctTxt)&&map[+acctTxt]){ code=+acctTxt; how='코드 일치'; }
    else{
      const hit=Object.values(map).find(a=>a.name===acctTxt);
      if(hit){ code=hit.code; how='이름 일치'; }
      else{
        const inc=Object.values(map).find(a=>acctTxt&&a.name.indexOf(acctTxt)>=0);
        if(inc){ code=inc.code; how='이름 포함(확인)'; }

      }

    rows.push({date:date,memo:memo,acctTxt:acctTxt,code:code,how:how,p:p,d:d,c:cr});
  });
  return {ok:true,rows:rows};
}
